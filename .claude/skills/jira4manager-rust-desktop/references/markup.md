## `ui/app.slint`

```slint
import { Button, LineEdit, ListView, VerticalBox } from "std-widgets.slint";
import { TaskItem, TaskRow } from "task-row.slint";
import { Tokens } from "tokens.slint";

export { TaskItem }

export component AppWindow inherits Window {
    in property <[TaskItem]> tasks;
    in property <string> error;
    in property <bool> busy;
    callback add(title: string);
    callback complete(id: string);
    callback remove(id: string);

    title: "Tally";
    preferred-width: 420px;
    preferred-height: 480px;

    VerticalBox {
        spacing: Tokens.space-m;

        HorizontalLayout {
            spacing: Tokens.space-s;

            title-input := LineEdit {
                placeholder-text: "New task";
                accessible-label: "New task title";
                accepted(text) => {
                    root.add(text);
                    self.text = "";
                }
            }

            Button {
                text: "Add";
                enabled: !root.busy;
                clicked => {
                    root.add(title-input.text);
                    title-input.text = "";
                }
            }
        }

        if root.error != "": Text {
            text: root.error;
            color: Tokens.danger;
            font-size: Tokens.text-body;
            accessible-role: text;
            accessible-label: "Error: \{root.error}";
        }

        ListView {
            for task in root.tasks: TaskRow {
                task: task;
                complete => {
                    root.complete(task.id);
                }
                remove => {
                    root.remove(task.id);
                }
            }
        }
    }
}
```

`task-row.slint` declares the `TaskItem` struct beside the component that shows it. `app.slint` re-exports the struct, so the build generates `TaskItem` for Rust.

## `src/view.rs` render

```rust
pub fn render(ui: &AppWindow, state: ViewState) {
    let items: Vec<TaskItem> = state
        .rows
        .into_iter()
        .map(|row| TaskItem {
            id: row.id.into(),
            title: row.title.into(),
            done: row.done,
        })
        .collect();
    ui.set_tasks(ModelRc::new(VecModel::from(items)));
    ui.set_error(state.error.unwrap_or_default().into());
    ui.set_busy(false);
}
```

## `src/main.rs`

```rust
use std::path::PathBuf;
use std::sync::Arc;

use slint::ComponentHandle;
use tally_core::tasks::TaskService;
use tally_desktop::{AppWindow, wire};
use tally_infra::{JsonFileTaskRepository, SystemClock};

fn main() -> anyhow::Result<()> {
    let runtime = tokio::runtime::Builder::new_multi_thread()
        .worker_threads(2)
        .enable_all()
        .build()?;
    let data_file =
        std::env::var_os("TALLY_DATA_FILE").map_or_else(|| "tally.json".into(), PathBuf::from);
    let tasks = TaskService::new(
        Arc::new(JsonFileTaskRepository::new(data_file)),
        Arc::new(SystemClock),
    );

    let ui = AppWindow::new()?;
    wire(&ui, tasks, runtime.handle().clone());
    ui.run()?;
    Ok(())
}
```

The desktop host composes in `main.rs`, because its graph is one service. Move the wiring to `compose.rs` when a second service arrives.
